## Implementation Details

### Step 1: Create thinking.tsx
```tsx
"use client";

import { cn } from "@/lib/utils";
import { SparklesIcon } from "lucide-react";
import type { HTMLAttributes } from "react";

export type ThinkingMessageProps = HTMLAttributes<HTMLDivElement>;

export const ThinkingMessage = ({
  className,
  ...props
}: ThinkingMessageProps) => {
  return (
    <div
      className={cn(
        "group/message fade-in w-full animate-in duration-300",
        className
      )}
      data-role="assistant"
      {...props}
    >
      <div className="flex items-start justify-start gap-3">
        {/* Sparkle icon with pulse animation */}
        <div className="-mt-1 flex size-8 shrink-0 items-center justify-center rounded-full bg-background ring-1 ring-border">
          <div className="animate-pulse">
            <SparklesIcon className="size-4" />
          </div>
        </div>

        {/* Thinking text with bouncing dots */}
        <div className="flex w-full flex-col gap-2">
          <div className="flex items-center gap-1 p-0 text-muted-foreground text-sm">
            <span className="animate-pulse">Thinking</span>
            <span className="inline-flex">
              <span className="animate-bounce [animation-delay:0ms]">.</span>
              <span className="animate-bounce [animation-delay:150ms]">.</span>
              <span className="animate-bounce [animation-delay:300ms]">.</span>
            </span>
          </div>
        </div>
      </div>
    </div>
  );
};
```

## Styling Notes
- The sparkle icon should have a subtle pulse animation
- The dots should bounce with staggered delays (0ms, 150ms, 300ms)
- Use Tailwind's built-in `animate-pulse` and `animate-bounce` utilities
- The component should match the assistant message layout (left-aligned with icon)

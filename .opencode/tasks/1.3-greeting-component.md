## Implementation Details

### Step 1: Create greeting.tsx
```tsx
"use client";

import { cn } from "@/lib/utils";
import { motion } from "framer-motion";
import type { HTMLAttributes } from "react";

export type GreetingProps = HTMLAttributes<HTMLDivElement>;

export const Greeting = ({ className, ...props }: GreetingProps) => {
  return (
    <div
      className={cn(
        "flex w-full flex-col items-center justify-center gap-4 px-4 py-8 text-center",
        className
      )}
      {...props}
    >
      <motion.div
        initial={{ opacity: 0, y: 10 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.5, ease: "easeOut" }}
        className="flex flex-col gap-2"
      >
        <h2 className="text-2xl font-semibold text-foreground">
          Hello there!
        </h2>
        <p className="text-muted-foreground">
          How can I help you today?
        </p>
      </motion.div>
    </div>
  );
};

// Alternative without framer-motion (using CSS animations)
export const GreetingSimple = ({ className, ...props }: GreetingProps) => {
  return (
    <div
      className={cn(
        "flex w-full flex-col items-center justify-center gap-4 px-4 py-8 text-center",
        "animate-in fade-in slide-in-from-bottom-2 duration-500",
        className
      )}
      {...props}
    >
      <div className="flex flex-col gap-2">
        <h2 className="text-2xl font-semibold text-foreground">
          Hello there!
        </h2>
        <p className="text-muted-foreground">
          How can I help you today?
        </p>
      </div>
    </div>
  );
};
```

## Styling Notes
- Center the greeting vertically and horizontally in the conversation area
- Use subtle fade-in and slide-up animation
- Text should be friendly but professional
- Consider using the app's primary color for emphasis (optional)

## Framer Motion Note
If framer-motion is not installed or you want to avoid adding dependencies, use the `GreetingSimple` version with Tailwind CSS animations (`animate-in` from tailwindcss-animate).

Check if framer-motion is available:
```bash
grep "framer-motion" package.json
```

If not installed, use the CSS-only version or install it:
```bash
bun add framer-motion
```

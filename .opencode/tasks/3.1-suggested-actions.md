## Implementation Details

### Step 1: Create suggested-actions.tsx

```tsx
"use client";

import { Button } from "@/components/ui/button";
import { cn } from "@/lib/utils";
import { motion } from "framer-motion";
import type { ComponentProps } from "react";

export type SuggestedAction = {
  title: string;
  description: string;
  prompt: string;
};

// Default suggestions - can be customized
export const defaultSuggestions: SuggestedAction[] = [
  {
    title: "Explain a concept",
    description: "Break down complex topics simply",
    prompt: "Explain quantum computing in simple terms",
  },
  {
    title: "Write code",
    description: "Help with programming tasks",
    prompt: "Write a Python function to sort a list of dictionaries by a key",
  },
  {
    title: "Brainstorm ideas",
    description: "Generate creative suggestions",
    prompt: "Give me 5 unique project ideas for learning React",
  },
  {
    title: "Summarize text",
    description: "Condense long content",
    prompt: "Summarize the key points of this article: ",
  },
];

export type SuggestedActionsProps = ComponentProps<"div"> & {
  suggestions?: SuggestedAction[];
  onSelect: (prompt: string) => void;
};

export const SuggestedActions = ({
  className,
  suggestions = defaultSuggestions,
  onSelect,
  ...props
}: SuggestedActionsProps) => {
  return (
    <div
      className={cn(
        "grid w-full gap-2 sm:grid-cols-2",
        className
      )}
      {...props}
    >
      {suggestions.map((suggestion, index) => (
        <motion.div
          key={suggestion.title}
          initial={{ opacity: 0, y: 10 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ 
            duration: 0.3, 
            delay: index * 0.1,
            ease: "easeOut" 
          }}
        >
          <SuggestedActionButton
            suggestion={suggestion}
            onClick={() => onSelect(suggestion.prompt)}
          />
        </motion.div>
      ))}
    </div>
  );
};

// CSS-only version without framer-motion
export const SuggestedActionsSimple = ({
  className,
  suggestions = defaultSuggestions,
  onSelect,
  ...props
}: SuggestedActionsProps) => {
  return (
    <div
      className={cn(
        "grid w-full gap-2 sm:grid-cols-2",
        className
      )}
      {...props}
    >
      {suggestions.map((suggestion, index) => (
        <div
          key={suggestion.title}
          className="animate-in fade-in slide-in-from-bottom-2"
          style={{ animationDelay: `${index * 100}ms` }}
        >
          <SuggestedActionButton
            suggestion={suggestion}
            onClick={() => onSelect(suggestion.prompt)}
          />
        </div>
      ))}
    </div>
  );
};

type SuggestedActionButtonProps = {
  suggestion: SuggestedAction;
  onClick: () => void;
};

const SuggestedActionButton = ({
  suggestion,
  onClick,
}: SuggestedActionButtonProps) => (
  <Button
    variant="outline"
    className="h-auto w-full flex-col items-start gap-1 p-4 text-left hover:bg-accent"
    onClick={onClick}
  >
    <span className="font-medium">{suggestion.title}</span>
    <span className="text-xs text-muted-foreground">
      {suggestion.description}
    </span>
  </Button>
);
```

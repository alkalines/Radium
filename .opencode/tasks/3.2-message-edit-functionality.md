### Step 1: Create message-editor.tsx

```tsx
"use client";

import { Button } from "@/components/ui/button";
import { Textarea } from "@/components/ui/textarea";
import { cn } from "@/lib/utils";
import type { UIMessage } from "ai";
import { useEffect, useRef, useState } from "react";

export type MessageEditorProps = {
  message: UIMessage;
  onSave: (newText: string) => void;
  onCancel: () => void;
  className?: string;
};

export const MessageEditor = ({
  message,
  onSave,
  onCancel,
  className,
}: MessageEditorProps) => {
  // Get the text content from the message
  const originalText = message.parts
    .filter((part) => part.type === "text")
    .map((part) => part.text)
    .join("\n");

  const [editedText, setEditedText] = useState(originalText);
  const textareaRef = useRef<HTMLTextAreaElement>(null);

  // Focus textarea on mount
  useEffect(() => {
    if (textareaRef.current) {
      textareaRef.current.focus();
      // Move cursor to end
      textareaRef.current.setSelectionRange(
        editedText.length,
        editedText.length
      );
    }
  }, [editedText.length]);

  const handleSave = () => {
    const trimmed = editedText.trim();
    if (trimmed && trimmed !== originalText) {
      onSave(trimmed);
    } else {
      onCancel();
    }
  };

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === "Escape") {
      onCancel();
    }
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault();
      handleSave();
    }
  };

  return (
    <div className={cn("flex flex-col gap-2 w-full", className)}>
      <Textarea
        ref={textareaRef}
        value={editedText}
        onChange={(e) => setEditedText(e.target.value)}
        onKeyDown={handleKeyDown}
        className="min-h-[80px] resize-none"
        placeholder="Edit your message..."
      />
      <div className="flex justify-end gap-2">
        <Button
          variant="ghost"
          size="sm"
          onClick={onCancel}
        >
          Cancel
        </Button>
        <Button
          size="sm"
          onClick={handleSave}
          disabled={!editedText.trim() || editedText.trim() === originalText}
        >
          Save & Regenerate
        </Button>
      </div>
    </div>
  );
};
```

### Step 2: Add Edit State Management to page.tsx

```tsx
// Add state for tracking which message is being edited
const [editingMessageId, setEditingMessageId] = useState<string | null>(null);

// Handler for starting edit
const handleStartEdit = (messageId: string) => {
  setEditingMessageId(messageId);
};

// Handler for canceling edit
const handleCancelEdit = () => {
  setEditingMessageId(null);
};

// Handler for saving edit
const handleSaveEdit = async (messageId: string, newText: string) => {
  // Find the index of the message being edited
  const messageIndex = messages.findIndex((m) => m.id === messageId);
  if (messageIndex === -1) return;

  // Delete all messages after this one (trailing messages)
  const messagesToKeep = messages.slice(0, messageIndex);
  
  // Update the edited message
  const editedMessage: UIMessage = {
    ...messages[messageIndex],
    parts: [{ type: "text", text: newText }],
  };

  // Set the new messages array
  setMessages([...messagesToKeep, editedMessage]);
  
  // Clear edit mode
  setEditingMessageId(null);

  // Regenerate response from the edited message
  // The AI SDK will handle this when we send the message again
  sendMessage(
    { text: newText },
    {
      body: {
        model: models?.find((m) => m._id === selectedModel)?.slug,
      },
    }
  );
};
```

## Alternative: Using AI SDK's setMessages Directly

The AI SDK provides `setMessages` which can be used for more complex editing scenarios:

```tsx
const handleSaveEdit = (messageId: string, newText: string) => {
  setMessages((currentMessages) => {
    const messageIndex = currentMessages.findIndex((m) => m.id === messageId);
    if (messageIndex === -1) return currentMessages;

    // Keep messages up to and including the edited one
    const newMessages = currentMessages.slice(0, messageIndex + 1);
    
    // Update the edited message's content
    newMessages[messageIndex] = {
      ...newMessages[messageIndex],
      parts: [{ type: "text", text: newText }],
    };

    return newMessages;
  });

  // Then send to regenerate
  sendMessage({ text: newText }, { body: { model: ... } });
};
```

### Step 1: Create a copy utility (if not exists)
```tsx
// In page.tsx or create utils/clipboard.ts
const copyToClipboard = async (text: string) => {
  try {
    await navigator.clipboard.writeText(text);
    return true;
  } catch (err) {
    console.error("Failed to copy:", err);
    return false;
  }
};
```

### Step 3: Create CopyMessageAction component

You can either create this inline or as a separate component:

```tsx
// Option A: Inline in page.tsx
const CopyMessageAction = ({ text }: { text: string }) => {
  const [copied, setCopied] = useState(false);
  
  const handleCopy = async () => {
    const success = await copyToClipboard(text);
    if (success) {
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    }
  };
  
  return (
    <MessageActions className="opacity-0 group-hover:opacity-100 transition-opacity">
      <MessageAction
        onClick={handleCopy}
        tooltip={copied ? "Copied!" : "Copy message"}
      >
        {copied ? (
          <CheckIcon className="size-4 text-green-500" />
        ) : (
          <CopyIcon className="size-4" />
        )}
      </MessageAction>
    </MessageActions>
  );
};
```

```tsx
// Option B: Add to message.tsx as a reusable component
export type CopyActionProps = {
  text: string;
  className?: string;
};

export const CopyAction = ({ text, className }: CopyActionProps) => {
  const [copied, setCopied] = useState(false);
  
  const handleCopy = async () => {
    try {
      await navigator.clipboard.writeText(text);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    } catch (err) {
      console.error("Failed to copy:", err);
    }
  };
  
  return (
    <MessageAction
      onClick={handleCopy}
      tooltip={copied ? "Copied!" : "Copy"}
      className={className}
    >
      {copied ? (
        <CheckIcon className="size-4 text-green-500" />
      ) : (
        <CopyIcon className="size-4" />
      )}
    </MessageAction>
  );
};
```

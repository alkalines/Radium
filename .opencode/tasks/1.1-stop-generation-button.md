## Goal
Allow users to stop AI responses mid-stream by clicking the submit button while streaming.

## Current State

### Step 3: Update PromptInputSubmit
```tsx
type PromptInputSubmitProps = ComponentProps<"button"> & {
  status: ChatStatus;
  onStop?: () => void;
};

export const PromptInputSubmit = ({
  status,
  onStop,
  disabled,
  className,
  ...props
}: PromptInputSubmitProps) => {
  const isStreaming = status === "streaming" || status === "submitted";
  
  const handleClick = (e: React.MouseEvent) => {
    if (isStreaming && onStop) {
      e.preventDefault();
      onStop();
    }
  };

  return (
    <Button
      type={isStreaming ? "button" : "submit"}
      disabled={isStreaming ? false : disabled}
      onClick={handleClick}
      className={cn(
        isStreaming && "bg-destructive hover:bg-destructive/90",
        className
      )}
      {...props}
    >
      {isStreaming ? <SquareIcon size={16} /> : <ArrowUpIcon size={16} />}
    </Button>
  );
};
```
